// ZPlugin: UDIM Baker Plugin for ZBrush 2026
[ISubPalette, "Zplugin:UDIM Baker"]

[VarDef, g_BridgeDll, ""]
[VarDef, g_CacheLow, "C:/Users/Public/Documents/ZBrushData2026/ZPluginData/UDIMBaker/UDIM_LowPoly.obj"]
[VarDef, g_CacheHigh, "C:/Users/Public/Documents/ZBrushData2026/ZPluginData/UDIMBaker/UDIM_HighPoly.obj"]
[VarDef, g_OrigSDiv, 1]
[VarDef, g_MaxSDiv, 1]
[VarDef, g_Res, 0]

// Button 1: One-Click Bake Active (Auto SDiv 1 to Max)
[IButton, "Zplugin:UDIM Baker:Bake Active SDiv 1 to Max", "Auto-export SDiv 1 as Low-Poly and Max SDiv as High-Poly, then launch Baker",
    [VarSet, g_BridgeDll, [FileNameResolvePath, "ZSTARTUP_ZPLUGS64/UDIMBakerData/UDIMBakerBridge.dll"]]
    
    // Check if active SubTool has SDiv levels
    [If, [IsEnabled, "Tool:Geometry:SDiv"],
        [VarSet, g_OrigSDiv, [IGet, "Tool:Geometry:SDiv"]]
        [VarSet, g_MaxSDiv, [IGetMax, "Tool:Geometry:SDiv"]]

        // 1. Export SDiv 1 (Low-Poly with UVs)
        [ISet, "Tool:Geometry:SDiv", 1]
        [FileNameSetNext, [Var, g_CacheLow]]
        [IPress, "Tool:Export"]

        // 2. Export Highest SDiv (High-Poly Sculpt)
        [ISet, "Tool:Geometry:SDiv", [Var, g_MaxSDiv]]
        [FileNameSetNext, [Var, g_CacheHigh]]
        [IPress, "Tool:Export"]

        // Restore original SDiv level
        [ISet, "Tool:Geometry:SDiv", [Var, g_OrigSDiv]]
    ,
        // No SDiv: export current mesh as high poly
        [FileNameSetNext, [Var, g_CacheHigh]]
        [IPress, "Tool:Export"]
    ]

    // Launch Baker with both meshes pre-loaded
    [VarSet, g_Res, [FileExecute, [Var, g_BridgeDll], "LaunchBakerSession", [StrMerge, [Var, g_CacheHigh], "|", [Var, g_CacheLow]]]]
]

// Button 2: Set Active SubTool as Low-Poly Target
[IButton, "Zplugin:UDIM Baker:Set Active as Low-Poly", "Set the currently active SubTool as the target Low-Poly mesh",
    [VarSet, g_BridgeDll, [FileNameResolvePath, "ZSTARTUP_ZPLUGS64/UDIMBakerData/UDIMBakerBridge.dll"]]
    [If, [IsEnabled, "Tool:Geometry:SDiv"],
        [VarSet, g_OrigSDiv, [IGet, "Tool:Geometry:SDiv"]]
        [ISet, "Tool:Geometry:SDiv", 1]
        [FileNameSetNext, [Var, g_CacheLow]]
        [IPress, "Tool:Export"]
        [ISet, "Tool:Geometry:SDiv", [Var, g_OrigSDiv]]
    ,
        [FileNameSetNext, [Var, g_CacheLow]]
        [IPress, "Tool:Export"]
    ]
    [VarSet, g_Res, [FileExecute, [Var, g_BridgeDll], "LaunchBakerSession", [StrMerge, "|", [Var, g_CacheLow]]]]
    [Note, "Active SubTool saved as Low-Poly Target & synchronized with Baker!", , 2]
]

// Button 3: Set Active SubTool as High-Poly Source
[IButton, "Zplugin:UDIM Baker:Set Active as High-Poly", "Set the currently active SubTool as the source High-Poly sculpt",
    [VarSet, g_BridgeDll, [FileNameResolvePath, "ZSTARTUP_ZPLUGS64/UDIMBakerData/UDIMBakerBridge.dll"]]
    [If, [IsEnabled, "Tool:Geometry:SDiv"],
        [VarSet, g_OrigSDiv, [IGet, "Tool:Geometry:SDiv"]]
        [VarSet, g_MaxSDiv, [IGetMax, "Tool:Geometry:SDiv"]]
        [ISet, "Tool:Geometry:SDiv", [Var, g_MaxSDiv]]
        [FileNameSetNext, [Var, g_CacheHigh]]
        [IPress, "Tool:Export"]
        [ISet, "Tool:Geometry:SDiv", [Var, g_OrigSDiv]]
    ,
        [FileNameSetNext, [Var, g_CacheHigh]]
        [IPress, "Tool:Export"]
    ]
    [VarSet, g_Res, [FileExecute, [Var, g_BridgeDll], "LaunchBakerSession", [Var, g_CacheHigh]]]
    [Note, "Active SubTool saved as High-Poly Source & synchronized with Baker!", , 2]
]

// Button 4: Launch Companion Baker Viewport
[IButton, "Zplugin:UDIM Baker:Launch Baker Viewport", "Open the companion baking viewport with current session",
    [VarSet, g_BridgeDll, [FileNameResolvePath, "ZSTARTUP_ZPLUGS64/UDIMBakerData/UDIMBakerBridge.dll"]]
    [VarSet, g_Res, [FileExecute, [Var, g_BridgeDll], "LaunchBakerSession", ""]]
]
